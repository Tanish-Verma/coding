#include <stdio.h>
#include <math.h>

int factorial(int N) {
    int result = 1;
    for (int i = 2; i <= N; i++) {
        result *= i;
    }
    return result;
}

long long int func1(int x) {
    return 3 * x + x * x;
}

long long int func2(int x) {
    return 64 * x - 7 * x * x * x;
}

long long int modulus(long long int x, int k) {
    if (x < 0) return (k - (-x % k)) % k;
    return x % k;
}

long long int Hi(int x, int i, int k) {
    long long int temp = (long long int)pow(-1, i);
    return modulus(func1(i) * func2(x) + func2(i) * func1(x) + i * temp, k);
}

long long int HS(int m, int arr[m], int x, int k) {
    long long int y = Hi(x, arr[m - 1], k);
    for (int i = 2; i <= m; i++) {
        y = Hi(y, arr[m - i], k);
    }
    return y;
}

void swap(int arr[], int i, int j) {
    int temp = arr[i];
    arr[i] = arr[j];
    arr[j] = temp;
}

void generatepermutations(int M, int arr[M], int start, int stop, int result[][M], int *index) {
    if (start == stop) {
        for (int i = 0; i <= stop; i++) {
            result[*index][i] = arr[i];
        }
        (*index)++;
    } else {
        for (int i = start; i <= stop; i++) {
            swap(arr, start, i);
            generatepermutations(M, arr, start + 1, stop, result, index);
            swap(arr, start, i); 
        }
    }
}
int isLexicographical(int a[], int b[], int size) {
    for (int i = 0; i < size; i++) {
        if (a[i] < b[i]) return 1;
        if (a[i] > b[i]) return 0;
    }
    return 0;
}

int main() {
    int N, M, K;
    scanf("%d %d %d", &N, &M, &K);
    int D[N];
    for (int i = 0; i < N; i++) {
        scanf("%d", &D[i]);
    }
    int arr[M];
    for (int i = 0; i < M; i++) {
        arr[i] = i + 1;
    }
    int total_permutations = factorial(M);
    int result[total_permutations][M];
    int index = 0; 
    generatepermutations(M, arr, 0, M - 1, result, &index); 
    long long int s = 0;
    for (int i = 0; i < N; i++) {
        s += D[i];
        long long int min_value = HS(M, result[0], s, K);
        int minarray[M];
        for (int k = 0; k < M; k++) {
            minarray[k] = result[0][k];
        }
        for (int j = 1; j < total_permutations; j++) {
            long long int current_value = HS(M, result[j], s, K);
            if (current_value < min_value || (current_value == min_value && isLexicographical(result[j], minarray, M))) {
                min_value = current_value;
                for (int k = 0; k < M; k++) {
                    minarray[k] = result[j][k];
                }
            }
        }
        for (int k = 0; k < M; k++) {
            printf("%d ", minarray[k]);
        }
        printf("\n");
    }

    return 0;
}
